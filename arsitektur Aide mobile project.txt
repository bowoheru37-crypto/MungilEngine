    private void initializeUI() {
        glSurfaceView = findViewById(R.id.gl_surface_view);
        webRuntime = findViewById(R.id.web_runtime);
        rootLayout = findViewById(R.id.editor_container);

        setupWebRuntime();
        setupBottomNavigation();
        setupTimeline();
        setupTools();

    }

    private void setupGLSurfaceView() {
        if (glSurfaceView != null) {
            glSurfaceView.setEGLContextClientVersion(3);
            glSurfaceView.setRenderer(new EngineRenderer(this, engineManager));
            glSurfaceView.setRenderMode(GLSurfaceView.RENDERMODE_WHEN_DIRTY);
        }
    }

    private void setupWebRuntime() {
        if (webRuntime == null) return;
        WebSettings settings = webRuntime.getSettings();
        settings.setJavaScriptEnabled(true);
        settings.setDomStorageEnabled(true);
        settings.setAllowFileAccess(true);
        settings.setAllowContentAccess(true);
        settings.setMediaPlaybackRequiresUserGesture(false);
        settings.setCacheMode(WebSettings.LOAD_NO_CACHE);
        settings.setSupportZoom(true);
        settings.setBuiltInZoomControls(true);
        settings.setDisplayZoomControls(false);
        settings.setMixedContentMode(WebSettings.MIXED_CONTENT_ALWAYS_ALLOW);

        webRuntime.setWebChromeClient(new WebChromeClient());
        webRuntime.setWebViewClient(new WebViewClient());
        webRuntime.addJavascriptInterface(new WebInterface(this), "AndroidEngine");

        // Load WASM runtime HTML
        webRuntime.loadUrl("file:///android_asset/wasm_runtime.html");
    }

    private void setupBottomNavigation() {
        if (bottomNav == null) return;
        bottomNav.setOnItemSelectedListener(item -> {
            int id = item.getItemId();
            if (id == R.id.nav_home) {
                showHome();
                return true;
            } else if (id == R.id.nav_projects) {
                openProjects();
                return true;
            } else if (id == R.id.nav_assets) {
                openAssets();
                return true;
            } else if (id == R.id.nav_templates) {
                openTemplates();
                return true;
            } else if (id == R.id.nav_settings) {
                openSettings();
                return true;
            }
            return false;
        });
    }

    private void setupTimeline() {
        if (timelineSeekBar == null) return;
        timelineSeekBar.setMax(1000);
        timelineSeekBar.setOnSeekBarChangeListener(new SeekBar.OnSeekBarChangeListener() {
				@Override
				public void onProgressChanged(SeekBar seekBar, int progress, boolean fromUser) {
					if (fromUser) {
						currentTime = (progress / 1000f) * totalTime;
						updateTimecode();
					}
				}
				@Override
				public void onStartTrackingTouch(SeekBar seekBar) {}
				@Override
				public void onStopTrackingTouch(SeekBar seekBar) {}
			});

        if (btnTimelinePlay != null) {
            btnTimelinePlay.setOnClickListener(v -> togglePlayback());
        }
        updateTimecode();
    }

    private void setupTools() {
        Map<Integer, String> toolMap = new HashMap<>();
        toolMap.put(R.id.tool_select, "select");
        toolMap.put(R.id.tool_move, "move");
        toolMap.put(R.id.tool_rotate, "rotate");
        toolMap.put(R.id.tool_scale, "scale");
        toolMap.put(R.id.tool_transform, "transform");
        toolMap.put(R.id.tool_draw, "draw");
        toolMap.put(R.id.tool_text, "text");
        toolMap.put(R.id.tool_image, "image");
        toolMap.put(R.id.tool_shape, "shape");

        for (Map.Entry<Integer, String> entry : toolMap.entrySet()) {
            View v = findViewById(entry.getKey());
            if (v != null) {
                v.setOnClickListener(view -> {
                    engineManager.setActiveTool(entry.getValue());
                    updateToolSelection(entry.getKey());
                });
            }
        }
    }

    private void setupViewPager() {
        if (bottomPager == null || bottomTabs == null) return;
        // Setup ViewPager adapter for bottom tabs
        bottomTabs.addOnTabSelectedListener(new TabLayout.OnTabSelectedListener() {
				@Override public void onTabSelected(TabLayout.Tab tab) {
					bottomPager.setCurrentItem(tab.getPosition());
				}
				@Override public void onTabUnselected(TabLayout.Tab tab) {}
				@Override public void onTabReselected(TabLayout.Tab tab) {}
			});
    }

    private void setupFab() {

    }

    private void showAddObjectDialog() {

    }

    private void setupInteractions() {
        // Mode toggle
        View btn2d = findViewById(R.id.btn_mode_2d);
        View btn3d = findViewById(R.id.btn_mode_3d);
        if (btn2d != null) btn2d.setOnClickListener(v -> setMode2D());
        if (btn3d != null) btn3d.setOnClickListener(v -> setMode3D());
    }

    private void setMode2D() {
        if (glSurfaceView != null) glSurfaceView.setVisibility(View.VISIBLE);
        if (webRuntime != null) webRuntime.setVisibility(View.GONE);
        engineManager.setActiveMode("2d");
    }

    private void setMode3D() {
        if (glSurfaceView != null) glSurfaceView.setVisibility(View.VISIBLE);
        if (webRuntime != null) webRuntime.setVisibility(View.GONE);
        engineManager.setActiveMode("3d");
    }

    }

    private void loadDefaultScene() {
        Scene defaultScene = new Scene();
        defaultScene.setName("Default Scene");
        defaultScene.addObject(new GameObject("Tree", "nature", new float[]{0, 0, 0}));
        defaultScene.addObject(new GameObject("Rock", "nature", new float[]{2, 0, 1}));
        defaultScene.addObject(new GameObject("Grass", "nature", new float[]{-1, 0, 2}));
        defaultScene.addObject(new GameObject("House", "building", new float[]{0, 0, -2}));
        defaultScene.addObject(new GameObject("Water", "fluid", new float[]{1, -1, 1}));

        engineManager.loadScene(defaultScene);
        graphics3D.setScene(defaultScene);
    }

    private void togglePlayback() {
        boolean playing = isPlaying.getAndSet(!isPlaying.get());
        if (btnTimelinePlay != null) {
            btnTimelinePlay.setImageResource(playing ? R.drawable.ic_pause_small : R.drawable.ic_play_small);
        }
        if (!playing) {
            startPlaybackLoop();
        }
    }

    private void startPlaybackLoop() {
        executor.execute(() -> {
            while (isPlaying.get() && !isDestroyed) {
                currentTime += 0.016f;
                if (currentTime > totalTime) currentTime = 0;

                final float progress = (currentTime / totalTime) * 1000;
                runOnUiThread(() -> {
                    if (timelineSeekBar != null) timelineSeekBar.setProgress((int) progress);
                    updateTimecode();
                });

                animationEngine.update(currentTime);
                physicsEngine.step(0.016f);
                if (glSurfaceView != null) glSurfaceView.requestRender();

                try {
                    Thread.sleep(16);
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    break;
                }
            }
        });
    }

    private void updateTimecode() {
        if (tvTimecode == null) return;
        tvTimecode.setText(String.format("%02d:%02d.%02d / %02d:%02d.%02d",
										 (int)(currentTime / 60), (int)(currentTime % 60), (int)((currentTime % 1) * 100),
										 (int)(totalTime / 60), (int)(totalTime % 60), (int)((totalTime % 1) * 100)));
    }

    private void updateToolSelection(int activeId) {
        int[] tools = {R.id.tool_select, R.id.tool_move, R.id.tool_rotate, R.id.tool_scale,
			R.id.tool_transform, R.id.tool_draw, R.id.tool_text, R.id.tool_image, R.id.tool_shape};
        for (int tool : tools) {
            View v = findViewById(tool);
            if (v != null) {
                v.setBackgroundResource(tool == activeId ? R.drawable.bg_tool_selected : 
										android.R.attr.selectableItemBackgroundBorderless);
            }
        }
    }

    private void showHome() {
        Toast.makeText(this, "Home", Toast.LENGTH_SHORT).show();
    }

    private void openProjects() {

    }

    private void openAssets() { 
        startActivity(new Intent(this, AssetManagerActivity.class)); 
    }

    private void openTemplates() {
        Toast.makeText(this, "Templates", Toast.LENGTH_SHORT).show();
    }

    private void openSettings() { 
        startActivity(new Intent(this, SettingsActivity.class)); 
    }

    @Override
    protected void onResume() {
        super.onResume();
        if (glSurfaceView != null) glSurfaceView.onResume();
    }

    @Override
    protected void onPause() {
        super.onPause();
        if (glSurfaceView != null) glSurfaceView.onPause();
        isPlaying.set(false);
    }

    @Override
    protected void onDestroy() {
        super.onDestroy();
        isDestroyed = true;
        isPlaying.set(false);
        executor.shutdown();
        engineManager.shutdown();
    }
